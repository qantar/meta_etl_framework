.PHONY: help build test lint clean deploy stop logs status backup restore

# Default target
.DEFAULT_GOAL := help

# Variables
DOCKER_COMPOSE := docker-compose
DOCKER_COMPOSE_PROD := docker-compose -f docker-compose.yml -f docker-compose.prod.yml
PYTHON := python3
PIP := pip3
PROJECT_NAME := etl-framework
VERSION := $(shell cat VERSION || echo "1.0.0")

# Colors for output
RED := \033[31m
GREEN := \033[32m
YELLOW := \033[33m
BLUE := \033[34m
RESET := \033[0m

help: ## Show this help message
	@echo "$(BLUE)ETL Framework - Production Makefile$(RESET)"
	@echo "$(YELLOW)Version: $(VERSION)$(RESET)"
	@echo ""
	@echo "$(GREEN)Available targets:$(RESET)"
	@awk 'BEGIN {FS = ":.*?## "} /^[a-zA-Z_-]+:.*?## / {printf "  $(BLUE)%-20s$(RESET) %s\n", $$1, $$2}' $(MAKEFILE_LIST)

# ==========================================
# Database Management
# ==========================================

db-init: ## Initialize databases with sample data
	@echo "$(GREEN)Initializing databases...$(RESET)"
	$(DOCKER_COMPOSE) exec postgres-source psql -U etl_user -d source_db -f /docker-entrypoint-initdb.d/01-init.sql
	$(DOCKER_COMPOSE) exec postgres-target psql -U etl_user -d target_db -f /docker-entrypoint-initdb.d/01-init.sql
	$(DOCKER_COMPOSE) exec postgres-metadata psql -U etl_user -d metadata_db -f /docker-entrypoint-initdb.d/01-metadata-schema.sql
	@echo "$(GREEN)Databases initialized!$(RESET)"

db-reset: ## Reset all databases
	@echo "$(RED)Resetting all databases... This will delete all data!$(RESET)"
	@read -p "Are you sure? (y/N): " confirm && [ "$$confirm" = "y" ]
	$(DOCKER_COMPOSE) down -v
	docker volume prune -f
	$(DOCKER_COMPOSE) up -d postgres-source postgres-target postgres-metadata
	sleep 10
	make db-init
	@echo "$(GREEN)Databases reset complete!$(RESET)"

db-backup: ## Backup all databases
	@echo "$(GREEN)Backing up databases...$(RESET)"
	mkdir -p backups/$(shell date +%Y%m%d_%H%M%S)
	$(DOCKER_COMPOSE) exec postgres-source pg_dump -U etl_user source_db > backups/$(shell date +%Y%m%d_%H%M%S)/source_db.sql
	$(DOCKER_COMPOSE) exec postgres-target pg_dump -U etl_user target_db > backups/$(shell date +%Y%m%d_%H%M%S)/target_db.sql
	$(DOCKER_COMPOSE) exec postgres-metadata pg_dump -U etl_user metadata_db > backups/$(shell date +%Y%m%d_%H%M%S)/metadata_db.sql
	@echo "$(GREEN)Database backup complete!$(RESET)"

# ==========================================
# Monitoring & Maintenance
# ==========================================

cleanup: ## Clean up old data and optimize storage
	@echo "$(GREEN)Running cleanup and optimization...$(RESET)"
	$(DOCKER_COMPOSE) exec etl-python etl-cli maintenance cleanup --retention-days 30
	$(DOCKER_COMPOSE) exec etl-spark etl-cli maintenance optimize
	@echo "$(GREEN)Cleanup complete!$(RESET)"

backup: ## Create full system backup
	@echo "$(GREEN)Creating full system backup...$(RESET)"
	mkdir -p backups/full_$(shell date +%Y%m%d_%H%M%S)
	make db-backup
	$(DOCKER_COMPOSE) exec etl-python tar -czf /tmp/etl_data_backup.tar.gz /app/data
	docker cp $(shell docker-compose ps -q etl-python):/tmp/etl_data_backup.tar.gz backups/full_$(shell date +%Y%m%d_%H%M%S)/
	@echo "$(GREEN)Full system backup complete!$(RESET)"

restore: ## Restore from backup (requires BACKUP_DIR variable)
	@echo "$(RED)Restoring from backup...$(RESET)"
	@if [ -z "$(BACKUP_DIR)" ]; then echo "$(RED)Please specify BACKUP_DIR=path/to/backup$(RESET)"; exit 1; fi
	@read -p "This will overwrite current data. Continue? (y/N): " confirm && [ "$$confirm" = "y" ]
	make db-reset
	$(DOCKER_COMPOSE) exec postgres-source psql -U etl_user -d source_db < $(BACKUP_DIR)/source_db.sql
	$(DOCKER_COMPOSE) exec postgres-target psql -U etl_user -d target_db < $(BACKUP_DIR)/target_db.sql
	$(DOCKER_COMPOSE) exec postgres-metadata psql -U etl_user -d metadata_db < $(BACKUP_DIR)/metadata_db.sql
	@echo "$(GREEN)System restore complete!$(RESET)"

# ==========================================
# Cleanup
# ==========================================

clean: ## Clean up build artifacts and temporary files
	@echo "$(GREEN)Cleaning up...$(RESET)"
	find . -type f -name "*.pyc" -delete
	find . -type d -name "__pycache__" -delete
	find . -type d -name "*.egg-info" -exec rm -rf {} +
	rm -rf build/ dist/ htmlcov/ .coverage .pytest_cache/
	docker system prune -f
	@echo "$(GREEN)Cleanup complete!$(RESET)"

clean-all: clean ## Clean everything including Docker volumes
	@echo "$(RED)Cleaning everything including data volumes...$(RESET)"
	@read -p "This will delete all data! Continue? (y/N): " confirm && [ "$$confirm" = "y" ]
	$(DOCKER_COMPOSE) down -v --rmi all
	docker volume prune -f
	docker network prune -f
	@echo "$(GREEN)Complete cleanup finished!$(RESET)"
